unit Vulkan_Components_DataStore;

{------------------------------------------------------------------------------
  Vulkan_Components_DataStore.pas
  A flexible data store and Vulkan buffer manager for vertex and instance data.
  Designed for use with the Vulkan API and pasVulkan (github.com/BeRo1985/pasvulkan)
  This unit uses Vulkan.pas, PasVulkan.pas, and PasVulkan.Framework headers to interface with the Vulkan API.

------------------------------------------------------------------------------}

(*
A flexible, thread-safe data store and Vulkan buffer manager for vertex,
  instance, and index data. Designed for use with the Vulkan API and
  pasVulkan (github.com/BeRo1985/pasvulkan).

  OBJECT MODEL:
  - Object: Represents a renderable entity
    - Has ONE instance binding (for per-instance data)
    - Can have MULTIPLE vertex sets (different vertex buffers)
    - Can have MULTIPLE index sets (different index buffers)

  USAGE PATTERN:
    1. Create store: Store := TvgVulkanDataStore.Create;
    2. Initialize Vulkan: Store.InitializeVulkan(Device, Queue, Pool);
    3. Add object: ObjIdx := Store.AddObject;
    4. Add vertex set: VSetIdx := Store.AddObjectVertexSet(ObjIdx);
    5. Configure attributes: Store.AddVertexAttributes(Binding, Types, Locations);
    6. Allocate data: Store.Allocate(Binding, Count);
    7. Set data: Store.SetObjectVertexPosition(ObjIdx, VSetIdx, Idx, X, Y, Z);
    8. Transfer: Store.TransferToVulkan;

  Thread Safety: All public methods are thread-safe via critical sections.
*)

procedure BindVertexAndInstanceBuffers(DS: TvgVulkanDataStore; CmdBuffer: TvgCommandBuffer;
                                             VertexBinding, InstanceBinding: Cardinal; AFrameIndex: Integer);

var
  VB, IB: TpvVulkanBuffer;
  OffV, OffI: TVkDeviceSize;

  VBArray : Array of TVkBuffer;
  OFFArray: Array of TVkDeviceSize;
  VK_TransferToVertexArray  : Array Of TVkBufferMemoryBarrier  ;

  ArraySize, aCount:Integer;

  Procedure SetUpBarrier(aIndex:Integer; aBuffer: TpvVulkanBuffer;  aOffset:Integer);
  Begin
    (*   need to set up the renderpass correctly to use a Barrier

      VK_TransferToVertexArray[aIndex] := Default( TVkBufferMemoryBarrier);

      VK_TransferToVertexArray[aIndex].sType               := VK_STRUCTURE_TYPE_BUFFER_MEMORY_BARRIER;
      VK_TransferToVertexArray[aIndex].pNext               := nil;
      VK_TransferToVertexArray[aIndex].srcAccessMask       := TVkAccessFlags(VK_ACCESS_TRANSFER_WRITE_BIT);
      VK_TransferToVertexArray[aIndex].dstAccessMask       := TVkAccessFlags(VK_ACCESS_VERTEX_ATTRIBUTE_READ_BIT);
      VK_TransferToVertexArray[aIndex].srcQueueFamilyIndex := VK_QUEUE_FAMILY_IGNORED;
      VK_TransferToVertexArray[aIndex].dstQueueFamilyIndex := VK_QUEUE_FAMILY_IGNORED;
      VK_TransferToVertexArray[aIndex].buffer              := aBuffer.Handle;
      VK_TransferToVertexArray[aIndex].offset              := aOffset;
      VK_TransferToVertexArray[aIndex].size                := aBuffer.Size;
    *)
  End;

begin


  if not Assigned(DS) then
    raise EVulkanDataStoreException.Create('Data store cannot be nil');
  if not Assigned(CmdBuffer) then
    raise EVulkanDataStoreException.Create('Command buffer cannot be nil');

  ArraySize := 0  ;

  // Get vertex buffer and offset
  VB   := DS.GetVulkanBuffer(VertexBinding, AFrameIndex);
  OffV := DS.GetDeviceOffsetForBinding(VertexBinding, AFrameIndex);

  if Assigned(VB) and (VB.Handle<>VK_NULL_HANDLE) then
    Inc(ArraySize);

  // Get instance buffer and offset if valid
  if InstanceBinding <> INVALIDBINDING then
  begin
    IB   := DS.GetVulkanBuffer(InstanceBinding, AFrameIndex);
    OffI := DS.GetDeviceOffsetForBinding(InstanceBinding, AFrameIndex);
    If Assigned(IB) and (IB.Handle<>VK_NULL_HANDLE) then
       Inc(ArraySize);
  end
  else
  begin
    IB   := nil;
    OffI := 0;
  end;

  If arraySize>0 then
  Begin
    SetLength(VBArray,  ArraySize);
    SetLength(OFFArray, ArraySize);
    SetLength(VK_TransferToVertexArray, ArraySize);

    aCount := 0;

    If VertexBinding < InstanceBinding then
    Begin
        // Bind vertex buffer if assigned
        if Assigned(VB) and (VB.Handle<>VK_NULL_HANDLE) then
        Begin
           VBArray[aCount] := VB.Handle;
           OFFArray[aCount]:= OffV;
           SetUpBarrier(aCount, VB, OffV);
           inc(aCount);
        end;
        // Bind instance buffer if valid and assigned
        if Assigned(IB) and (IB.Handle<>VK_NULL_HANDLE) then
        Begin
           VBArray[aCount] := IB.Handle;
           OFFArray[aCount]:= OffI;
           SetUpBarrier(aCount, IB, OffI);
        end;


    end else
    Begin
        // Bind instance buffer if valid and assigned
        if Assigned(IB) and (IB.Handle<>VK_NULL_HANDLE) then
        Begin
           VBArray[aCount] := IB.Handle;
           OFFArray[aCount]:= OffI;
           SetUpBarrier(aCount, IB, OffI);
           inc(aCount);
        end;

        if Assigned(VB) and (VB.Handle<>VK_NULL_HANDLE) then
        Begin
           VBArray[aCount] := VB.Handle;
           OFFArray[aCount]:= OffV;
           SetUpBarrier(aCount, VB, OffV);
        end;

    End;

    CmdBuffer.CmdBindVertexBuffers(0, arraySize, @VBArray[0], @OFFArray[0]);

    SetLength(VBArray,0);
    SetLength(OFFArray,0);
  End;



end;

procedure DrawObject_PerBinding(DS: TvgVulkanDataStore; CmdBuffer: TvgCommandBuffer;
                                ObjectIndex: Integer;     AFrameIndex: Integer;
                                 VertexSetIndex: Integer = 0; IndexSetIndex: Integer = 0);

var
  VertexBinding, InstanceBinding, IndexBinding: Cardinal;
  VertexCount, InstanceCount, IndexCount: Integer;
  IndexBuffer: TpvVulkanBuffer;
  IndexOffset: TVkDeviceSize;
  IndexTypeVk: TVkIndexType;
begin
  if not Assigned(DS) then
    raise EVulkanDataStoreException.Create('Data store cannot be nil');
  if not Assigned(CmdBuffer) then
    raise EVulkanDataStoreException.Create('Command buffer cannot be nil');

  // Resolve bindings for this object (thread-safe via DS methods)
  VertexBinding   := DS.GetVertexBinding(ObjectIndex, VertexSetIndex);
  InstanceBinding := DS.GetInstanceBinding(ObjectIndex);
  IndexBinding    := DS.GetIndexBinding(ObjectIndex, IndexSetIndex);

  // Bind vertex + instance buffers (per-binding mode uses per-binding buffers, offsets = 0)
  BindVertexAndInstanceBuffers(DS, CmdBuffer, VertexBinding, InstanceBinding, AFrameIndex);

  // Bind index buffer - assumes per-binding uploads with offset 0
  IndexBuffer := GetVulkanIndexBufferForBinding(DS, IndexBinding, AFrameIndex);
  IndexOffset := 0;  // Per-binding mode: offsets are 0

  // Choose VkIndexType from DS.IndexType
  case DS.GetIndexType(IndexBinding) of
    itUInt16: IndexTypeVk := VK_INDEX_TYPE_UINT16;
    itUInt32: IndexTypeVk := VK_INDEX_TYPE_UINT32;
  else
    raise EVulkanDataStoreException.Create('Unsupported index type');
  end;

  // Bind index buffer if assigned
  if Assigned(IndexBuffer) then
    CmdBuffer.CmdBindIndexBuffer(IndexBuffer.Handle, IndexOffset, IndexTypeVk);

  // Get counts
  IndexCount      := DS.GetIndexCount(IndexBinding);
  VertexCount     := DS.GetCount(VertexBinding);
  if InstanceBinding <> INVALIDBINDING then
    InstanceCount := DS.GetCount(InstanceBinding)
  else
    InstanceCount := 1;

  // Issue draw: indexed if indices exist, else non-indexed
  if IndexCount > 0 then
    CmdBuffer.CmdDrawIndexed(IndexCount, InstanceCount, 0, 0, 0)
  else
    CmdBuffer.CmdDraw(VertexCount, InstanceCount, 0, 0);


end;

procedure DrawObject_SingleBlock(DS: TvgVulkanDataStore; CmdBuffer: TvgCommandBuffer;
  ObjectIndex: Integer; AFrameIndex: Integer; VertexSetIndex: Integer = 0; IndexSetIndex: Integer = 0);

var
  VertexBinding, InstanceBinding, IndexBinding: Cardinal;
  VertexCount, InstanceCount, IndexCount: Integer;
  UnifiedBuf, IndexBuf: TpvVulkanBuffer;
  VertexOffset, InstanceOffset, IndexOffset: TVkDeviceSize;
  IndexTypeVk: TVkIndexType;

begin
  if not Assigned(DS) then
    raise EVulkanDataStoreException.Create('Data store cannot be nil');
  if not Assigned(CmdBuffer) then
    raise EVulkanDataStoreException.Create('Command buffer cannot be nil');

  // Resolve bindings
  VertexBinding   := DS.GetVertexBinding(ObjectIndex, VertexSetIndex);
  InstanceBinding := DS.GetInstanceBinding(ObjectIndex);
  IndexBinding    := DS.GetIndexBinding(ObjectIndex, IndexSetIndex);

  // Unified buffer for vertex/instance (same buffer for all, with offsets)
  UnifiedBuf   := DS.GetVulkanBuffer(VertexBinding, AFrameIndex);  // All bindings share the unified buffer in this mode
  VertexOffset := DS.GetDeviceOffsetForBinding(VertexBinding, AFrameIndex);

  if InstanceBinding <> INVALIDBINDING then
    InstanceOffset := DS.GetDeviceOffsetForBinding(InstanceBinding, AFrameIndex)
  else
    InstanceOffset := 0;

  // Bind unified buffer to each binding slot with per-binding offsets
  if Assigned(UnifiedBuf) then
  begin
    CmdBuffer.CmdBindVertexBuffers(VertexBinding, 1, @UnifiedBuf.Handle, @VertexOffset);
    if InstanceBinding <> INVALIDBINDING then
      CmdBuffer.CmdBindVertexBuffers(InstanceBinding, 1, @UnifiedBuf.Handle, @InstanceOffset);
  end;

  // Index buffer (per-binding, not unified in the provided code)
  IndexBuf := GetVulkanIndexBufferForBinding(DS, IndexBinding, AFrameIndex);
  IndexOffset := 0;  // Indices are per-binding, not packed into unified buffer

  // Choose VkIndexType
  case DS.GetIndexType(IndexBinding) of
    itUInt16: IndexTypeVk := VK_INDEX_TYPE_UINT16;
    itUInt32: IndexTypeVk := VK_INDEX_TYPE_UINT32;
  else
    raise EVulkanDataStoreException.Create('Unsupported index type');
  end;

  // Bind index buffer if assigned
  if Assigned(IndexBuf) then
    CmdBuffer.CmdBindIndexBuffer(IndexBuf.Handle, IndexOffset, IndexTypeVk);

  // Get counts
  IndexCount := DS.GetIndexCount(IndexBinding);
  VertexCount := DS.GetCount(VertexBinding);
  if InstanceBinding <> INVALIDBINDING then
    InstanceCount := DS.GetCount(InstanceBinding)
  else
    InstanceCount := 1;

  // Issue draw
  if IndexCount > 0 then
    CmdBuffer.CmdDrawIndexed(IndexCount, InstanceCount, 0, 0, 0)
  else
    CmdBuffer.CmdDraw(VertexCount, InstanceCount, 0, 0);
end;

Procedure TvgVulkanDataStore.CreateVulkanDataBuffers;
var
  Binding: Cardinal;
  Frame: Integer;
  BufInfo: TVulkanBufferInfo;
  Data: TBytes;
  Size: TVkDeviceSize;
  IndexSize: TVkDeviceSize;
  TotalSize: TVkDeviceSize;
  Offset: TVkDeviceSize;
  Usage: TVkBufferUsageFlags;
  MemProps: TVkMemoryPropertyFlags;
  BindingBuffers: TArray<TVulkanBufferInfo>;
begin
  if fBuffersCreated then Exit; // Prevent recreation if already done

  CustomAssert(Assigned(FVulkanDevice), 'Vulkan device must be assigned before creating buffers');

  // Common flags

  Usage   := TVkBufferUsageFlags(VK_BUFFER_USAGE_VERTEX_BUFFER_BIT) or
             TVkBufferUsageFlags(VK_BUFFER_USAGE_TRANSFER_DST_BIT);

  MemProps:=TVkMemoryPropertyFlags(VK_MEMORY_PROPERTY_DEVICE_LOCAL_BIT);

//  Usage := VK_BUFFER_USAGE_TRANSFER_DST_BIT or VK_BUFFER_USAGE_VERTEX_BUFFER_BIT; // Adjust for instance if needed
//  MemProps := VK_MEMORY_PROPERTY_DEVICE_LOCAL_BIT;

  if FUploadMode = umSingleBlock then
  begin
    // Compute total size: Sum all vertex/instance + index data
    TotalSize := 0;
    FUnifiedOffsets.Clear;

    // Vertex/Instance data
    for Binding in FData.Keys do
    begin
      if (FData.TryGetValue(Binding, Data)) and
         (FDataSize.TryGetValue(Binding, Size)) then
      begin
       // Size := Length(Data); { MODIFIED: No longer fallback to Length(Data); rely on fDataSize for real used size }
        FUnifiedOffsets.Add(Binding, TotalSize);
        Inc(TotalSize, Size);
      end;
    end;

    // Index data
    for Binding in FIndexDatas.Keys do
    begin
      if FIndexDatas.TryGetValue(Binding, Data) and
        (FDataSize.TryGetValue(Binding, IndexSize)) then
      begin
        FUnifiedOffsets.Add(Binding, TotalSize); // Use same offsets dict, assume bindings don't overlap
        Inc(TotalSize, IndexSize);
      end;
    end;

    FUnifiedSize := TotalSize;
    SetLength(FUnifiedBuffer, FNumFrames);

    for Frame := 0 to FNumFrames - 1 do
    begin
      CreateVulkanBuffer(FVulkanDevice, FUnifiedSize,
                                      Usage + TVkBufferUsageFlags( VK_BUFFER_USAGE_INDEX_BUFFER_BIT),
                                      MemProps,
                                      FUnifiedBuffer[Frame]);
    end;
  end
  else // umPerBinding
  begin
    // Vertex/Instance buffers
    for Binding in FData.Keys do
    begin
      if not FData.TryGetValue(Binding, Data) then Continue;
      if not FDataSize.TryGetValue(Binding, Size) then Continue; { MODIFIED: Rely on fDataSize }

      SetLength(BindingBuffers, FNumFrames);
      for Frame := 0 to FNumFrames - 1 do
      begin
        CreateVulkanBuffer(FVulkanDevice, Size, Usage, MemProps, BufInfo);
        BindingBuffers[Frame] := BufInfo;
      end;
      FVulkanBuffers.AddOrSetValue(Binding, BindingBuffers);
    end;

    // Index buffers
    for Binding in FIndexDatas.Keys do
    begin
      if not FIndexDatas.TryGetValue(Binding, Data) then Continue;
      if not FDataSize.TryGetValue(Binding, IndexSize) then Continue; { MODIFIED: Rely on fDataSize }

      SetLength(BindingBuffers, FNumFrames);
      for Frame := 0 to FNumFrames - 1 do
      begin
        CreateVulkanBuffer(FVulkanDevice, IndexSize, TVkBufferUsageFlags(VK_BUFFER_USAGE_TRANSFER_DST_BIT) +
                                                     TVkBufferUsageFlags(VK_BUFFER_USAGE_INDEX_BUFFER_BIT), MemProps, BufInfo);
        BindingBuffers[Frame] := BufInfo;
      end;
      FIndexBuffers.AddOrSetValue(Binding, BindingBuffers);
    end;
  end;

  fBuffersCreated := True;
end;

Procedure TvgVulkanDataStore.DeleteVulkanDataBuffers;
var
  Binding: Cardinal;
  Frame: Integer;
  BindingBuffers: TArray<TVulkanBufferInfo>;
begin
  if not fBuffersCreated then Exit;

  if FUploadMode = umSingleBlock then
  begin
    for Frame := 0 to High(FUnifiedBuffer) do
      DestroyVulkanBuffer(FUnifiedBuffer[Frame]);
    SetLength(FUnifiedBuffer, 0);
    FUnifiedOffsets.Clear;
    FUnifiedSize := 0;
  end
  else // umPerBinding
  begin
    // Vertex/Instance buffers
    for Binding in FVulkanBuffers.Keys do
    begin
      if FVulkanBuffers.TryGetValue(Binding, BindingBuffers) then
      begin
        for Frame := 0 to High(BindingBuffers) do
          DestroyVulkanBuffer(BindingBuffers[Frame]);
        SetLength(BindingBuffers, 0);
      end;
    end;
    FVulkanBuffers.Clear;

    // Index buffers
    for Binding in FIndexBuffers.Keys do
    begin
      if FIndexBuffers.TryGetValue(Binding, BindingBuffers) then
      begin
        for Frame := 0 to High(BindingBuffers) do
          DestroyVulkanBuffer(BindingBuffers[Frame]);
        SetLength(BindingBuffers, 0);
      end;
    end;
    FIndexBuffers.Clear;
  end;

  fBuffersCreated := False;
end;

procedure TvgVulkanDataStore.Allocate(ABinding: Cardinal; ACount: Integer; AMode: TAllocationMode = amClear);
var
  Stride: Cardinal;
  Data: TBytes;
  RequiredBytes: NativeInt;
  CurrentCapacity: Integer;
begin
  FCriticalSection.Enter;
  try
    if ACount < 0 then
      raise EVulkanDataStoreException.CreateFmt('Count cannot be negative: %d', [ACount]);

    if not ValidateBinding(ABinding, 'Allocate') then Exit;

    Stride := GetStride(ABinding);

    // Ensure bookkeeping entries exist { MODIFIED: Ensure fDataSize exists too }
    if not FCapacities.ContainsKey(ABinding) then
      FCapacities.Add(ABinding, 0);
    if not FData.ContainsKey(ABinding) then
      FData.Add(ABinding, nil);
    if not FCounts.ContainsKey(ABinding) then
      FCounts.Add(ABinding, 0);
    if not FDataSize.ContainsKey(ABinding) then
      FDataSize.Add(ABinding, 0);

    CurrentCapacity := FCapacities[ABinding];

    RequiredBytes := NativeInt(ACount) * NativeInt(Stride);

    // If preserving and new count < current, just update count/size (trim later if needed)
    if (AMode = amPreserve) and (ACount < CurrentCapacity) then
    begin
      FCounts[ABinding] := ACount;
      FDataSize[ABinding] := TVkDeviceSize(RequiredBytes); { ADDED: Update fDataSize to new used bytes }
      SetDataDirty;
      Exit;
    end;

    // Ensure capacity (grows if needed)
    EnsureCapacity(ABinding, ACount);

    Data := FData[ABinding];

    case AMode of
      amClear:
        FillChar(Data[0], RequiredBytes, 0);
      amPreserve:
        // If growing, clear new area
        if ACount > CurrentCapacity then
          FillChar(Data[CurrentCapacity * Stride], (ACount - CurrentCapacity) * Stride, 0);
      amUninitialized:
        // No action; garbage OK
    end;

    FData[ABinding] := Data;
    FCounts[ABinding] := ACount;
    FDataSize[ABinding] := TVkDeviceSize(RequiredBytes); { ADDED: Set fDataSize to used bytes (real data) }
    SetDataDirty;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.EnsureCapacity(ABinding: Cardinal; ARequiredCount: Integer);
var
  Stride, NewCapacity, CurrentCapacity: Integer;
  Data: TBytes;
begin
  FCriticalSection.Enter;
  try
    if not FCapacities.TryGetValue(ABinding, CurrentCapacity) then
      CurrentCapacity := 0;

    if ARequiredCount <= CurrentCapacity then Exit;

    Stride := GetStride(ABinding);

    NewCapacity := Max(MIN_CAPACITY, Max(ARequiredCount, CurrentCapacity * 2)); // Power-of-2 growth

    Data := FData[ABinding];
    SetLength(Data, NewCapacity * Stride);

    FData[ABinding] := Data;
    FCapacities[ABinding] := NewCapacity;
  finally
    FCriticalSection.Leave;
  end;
end;

function TvgVulkanDataStore.AddVertex(ABinding: Cardinal): Integer;
var
  Count: Integer;
  Stride: Cardinal;
begin
  FCriticalSection.Enter;
  try
    if not FCounts.TryGetValue(ABinding, Count) then
      Count := 0;

    EnsureCapacity(ABinding, Count + 1);

    Inc(Count);
    FCounts[ABinding] := Count;

    Stride := GetStride(ABinding);
    FDataSize[ABinding] := TVkDeviceSize(Count) * Stride; { ADDED: Update fDataSize to new used bytes }

    SetDataDirty;
    Result := Count - 1;
  finally
    FCriticalSection.Leave;
  end;
end;

function TvgVulkanDataStore.AddVertices(ABinding: Cardinal; ACount: Integer): Integer;
var
  OldCount: Integer;
  Stride: Cardinal;
begin
  FCriticalSection.Enter;
  try
    if ACount <= 0 then Exit(0);

    if not FCounts.TryGetValue(ABinding, OldCount) then
      OldCount := 0;

    EnsureCapacity(ABinding, OldCount + ACount);

    Inc(OldCount, ACount);
    FCounts[ABinding] := OldCount;

    Stride := GetStride(ABinding);
    FDataSize[ABinding] := TVkDeviceSize(OldCount) * Stride; { ADDED: Update fDataSize to new used bytes }

    SetDataDirty;
    Result := OldCount - ACount; // Starting index of added vertices
  finally
    FCriticalSection.Leave;
  end;
end;

function TvgVulkanDataStore.SetCount(ABinding: Cardinal; ANewUsedCount: Integer): Cardinal;
var
  Stride: Integer;
begin
  FCriticalSection.Enter;
  try
    if ANewUsedCount < 0 then
      raise EVulkanDataStoreException.CreateFmt('Count cannot be negative: %d', [ANewUsedCount]);

    if not FCapacities.ContainsKey(ABinding) then
      raise EVulkanDataStoreException.CreateFmt('Binding %d not allocated', [ABinding]);

    if ANewUsedCount > FCapacities[ABinding] then
      EnsureCapacity(ABinding, ANewUsedCount); { MODIFIED: Grow capacity if needed }

    FCounts[ABinding] := ANewUsedCount;

    Stride := GetStride(ABinding);
    FDataSize[ABinding] := TVkDeviceSize(ANewUsedCount) * Stride; { ADDED: Update fDataSize to new used bytes }

    SetDataDirty;
    Result := ANewUsedCount;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.TrimExcess(ABinding: Cardinal);
var
  Count: Integer;
  Stride: Cardinal;
  Data: TBytes;
begin
  FCriticalSection.Enter;
  try
    if not FCounts.TryGetValue(ABinding, Count) then Exit;

    Stride := GetStride(ABinding);
    Data := FData[ABinding];
    SetLength(Data, Count * Stride);

    FData[ABinding] := Data;
    FCapacities[ABinding] := Count;
    // fDataSize remains unchanged (already Count * Stride)
    SetDataDirty;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.InitializeBindingIndex(ABinding: Cardinal);
var
  Data: TBytes;
begin
  FCriticalSection.Enter;
  try
    // If already initialized, do nothing
    if FIndexDatas.ContainsKey(ABinding) then
      Exit;

    SetLength(Data, 0); // Initialize as empty array
    FIndexDatas.AddOrSetValue(ABinding, Data);
    FIndexCounts.AddOrSetValue(ABinding, 0);
    FIndexCapacities.AddOrSetValue(ABinding, 0);
    FIndexTypes.AddOrSetValue(ABinding, itUInt32); // Default
    FDataSize.AddOrSetValue(ABinding, 0); { ADDED: Initialize fDataSize for index binding }
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.AllocateIndices(ABinding: Cardinal; ACount: Integer);
var
  IndexSize: Cardinal;
  Data: TBytes;
  RequiredBytes: NativeInt;
begin
  FCriticalSection.Enter;
  try
    if ACount < 0 then
      raise EVulkanDataStoreException.CreateFmt('Index count cannot be negative: %d', [ACount]);

    IndexSize := GetIndexElementSize(GetIndexType(ABinding));

    // Ensure bookkeeping entries exist { MODIFIED: Ensure fDataSize exists }
    if not FIndexCapacities.ContainsKey(ABinding) then
      FIndexCapacities.AddOrSetValue(ABinding, 0);
    if not FIndexDatas.ContainsKey(ABinding) then
      FIndexDatas.AddOrSetValue(ABinding, nil);
    if not FIndexCounts.ContainsKey(ABinding) then
      FIndexCounts.AddOrSetValue(ABinding, 0); // used count remains unchanged
    if not FDataSize.ContainsKey(ABinding) then
      FDataSize.AddOrSetValue(ABinding, 0);

    // Ensure capacity for indices (in elements)
    EnsureIndexCapacity(ABinding, ACount);

    // Optionally clear the first ACount indices (but do not alter FIndexCounts or fDataSize here; that's for add/set)
    Data := FIndexDatas[ABinding];
    RequiredBytes := NativeInt(ACount) * NativeInt(IndexSize);
    if RequiredBytes > 0 then
    begin
      if Length(Data) < RequiredBytes then
        SetLength(Data, RequiredBytes);
      FillChar(Data[0], RequiredBytes, 0);
      FIndexDatas[ABinding] := Data;
    end;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.EnsureIndexCapacity(ABinding: Cardinal; ARequiredCount: Integer);
var
  IndexSize, NewCapacity, CurrentCapacity: Integer;
  Data: TBytes;
begin
  FCriticalSection.Enter;
  try
    if not FIndexCapacities.TryGetValue(ABinding, CurrentCapacity) then
      CurrentCapacity := 0;

    if ARequiredCount <= CurrentCapacity then Exit;

    IndexSize := GetIndexElementSize(GetIndexType(ABinding));

    NewCapacity := Max(MIN_CAPACITY, Max(ARequiredCount, CurrentCapacity * 2));

    Data := FIndexDatas[ABinding];
    SetLength(Data, NewCapacity * IndexSize);

    FIndexDatas[ABinding] := Data;
    FIndexCapacities[ABinding] := NewCapacity;
  finally
    FCriticalSection.Leave;
  end;
end;

function TvgVulkanDataStore.AddIndex(ABinding: Cardinal; AIndex: Cardinal): Integer;
var
  Count: Integer;
  ElementSize: Cardinal;
begin
  FCriticalSection.Enter;
  try
    if not FIndexCounts.TryGetValue(ABinding, Count) then
      Count := 0;

    EnsureIndexCapacity(ABinding, Count + 1);
    SetIndex(ABinding, Count, AIndex);

    Inc(Count);
    FIndexCounts[ABinding] := Count;

    ElementSize := GetIndexElementSize(GetIndexType(ABinding));
    FDataSize[ABinding] := TVkDeviceSize(Count) * ElementSize; { ADDED: Update fDataSize to new used bytes for index }

    SetDataDirty;
    Result := Count - 1;
  finally
    FCriticalSection.Leave;
  end;
end;

function TvgVulkanDataStore.AddIndex(ABinding: Cardinal; I1, I2, I3: Cardinal): Integer;
begin
  Result := AddIndex(ABinding, I1);
  AddIndex(ABinding, I2);
  AddIndex(ABinding, I3);
  // fDataSize updated in each AddIndex call
end;

function TvgVulkanDataStore.AddIndices(ABinding: Cardinal; const AIndices: array of Cardinal): Integer;
var
  I, OldCount: Integer;
  ElementSize: Cardinal;
begin
  FCriticalSection.Enter;
  try
    Result := Length(AIndices);
    if Result = 0 then Exit;

    if not FIndexCounts.TryGetValue(ABinding, OldCount) then
      OldCount := 0;

    EnsureIndexCapacity(ABinding, OldCount + Result);

    for I := 0 to Result - 1 do
      SetIndex(ABinding, OldCount + I, AIndices[I]);

    Inc(OldCount, Result);
    FIndexCounts[ABinding] := OldCount;

    ElementSize := GetIndexElementSize(GetIndexType(ABinding));
    FDataSize[ABinding] := TVkDeviceSize(OldCount) * ElementSize; { ADDED: Update fDataSize to new used bytes for index }

    SetDataDirty;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.UploadAllData(aTransferPool: TvgCommandBufferPool; aFrameIndex: Integer);
var
  Binding: Cardinal;
  NeedRecreate: Boolean;
  BufferSize: TVkDeviceSize;
begin
  if not GetDataDirty(aFrameIndex) then Exit;

  FCriticalSection.Enter;
  try
    NeedRecreate := False;

    if FUploadMode = umPerBinding then
    begin
      // Check vertex/instance bindings
      for Binding in FVulkanBuffers.Keys do
      begin
        if not FDataSize.TryGetValue(Binding, BufferSize) then Continue;
        if (Length(FVulkanBuffers[Binding]) > 0) and (BufferSize > FVulkanBuffers[Binding][0].Size) then
        begin
          NeedRecreate := True;
          Break;
        end;
      end;

      // Check index bindings
      for Binding in FIndexBuffers.Keys do
      begin
        if not FDataSize.TryGetValue(Binding, BufferSize) then Continue;
        if (Length(FIndexBuffers[Binding]) > 0) and (BufferSize > FIndexBuffers[Binding][0].Size) then
        begin
          NeedRecreate := True;
          Break;
        end;
      end;
    end
    else // umSingleBlock
    begin
      // Compute current total used size
      BufferSize := 0;
      for Binding in FDataSize.Keys do
        Inc(BufferSize, FDataSize[Binding]);
      if BufferSize > FUnifiedSize then
        NeedRecreate := True;
    end;

    if NeedRecreate or not fBuffersCreated then
    begin
      DeleteVulkanDataBuffers;
      CreateVulkanDataBuffers;
    end;

    // Perform transfer
    if FUploadMode = umPerBinding then
      TransferToVulkan_PerBinding(aTransferPool, aFrameIndex)
    else
      TransferToVulkan_SingleBlock(aTransferPool, aFrameIndex);

    FDataDirty[aFrameIndex] := False;
  finally
    FCriticalSection.Leave;
  end;
end;

procedure TvgVulkanDataStore.TransferToVulkan_PerBinding(aTransferPool: TvgCommandBufferPool; aFrameIndex: Integer);
var
  Binding: Cardinal;
  Data: TBytes;
  Size: TVkDeviceSize;
  Buffer: TpvVulkanBuffer;
begin
  // Vertex/Instance data
  for Binding in FData.Keys do
  begin
    Data := FData[Binding];
    if not FDataSize.TryGetValue(Binding, Size) then Continue; { MODIFIED: Use fDataSize instead of Length(Data) }
    Buffer := FVulkanBuffers[Binding][aFrameIndex].Buffer;
    CopyBufferToDevice(aTransferPool, Data, Size, 0, aFrameIndex, Buffer);
  end;

  // Index data
  for Binding in FIndexDatas.Keys do
  begin
    Data := FIndexDatas[Binding];
    if not FDataSize.TryGetValue(Binding, Size) then Continue; { MODIFIED: Use fDataSize }
    Buffer := FIndexBuffers[Binding][aFrameIndex].Buffer;
    CopyBufferToDevice(aTransferPool, Data, Size, 0, aFrameIndex, Buffer);
  end;
end;

procedure TvgVulkanDataStore.TransferToVulkan_SingleBlock(aTransferPool: TvgCommandBufferPool; aFrameIndex: Integer);
var
  Binding: Cardinal;
  Data: TBytes;
  Size, Offset: TVkDeviceSize;
  UnifiedBuf: TpvVulkanBuffer;
begin
  UnifiedBuf := FUnifiedBuffer[aFrameIndex].Buffer;

  // Vertex/Instance data
  for Binding in FData.Keys do
  begin
    Data := FData[Binding];
    if not FDataSize.TryGetValue(Binding, Size) then Continue; { MODIFIED: Use fDataSize }
    if not FUnifiedOffsets.TryGetValue(Binding, Offset) then Continue;
    CopyBufferToDevice(aTransferPool, Data, Size, Offset, aFrameIndex, UnifiedBuf);
  end;

  // Index data
  for Binding in FIndexDatas.Keys do
  begin
    Data := FIndexDatas[Binding];
    if not FDataSize.TryGetValue(Binding, Size) then Continue; { MODIFIED: Use fDataSize }
    if not FUnifiedOffsets.TryGetValue(Binding, Offset) then Continue;
    CopyBufferToDevice(aTransferPool, Data, Size, Offset, aFrameIndex, UnifiedBuf);
  end;
end;
